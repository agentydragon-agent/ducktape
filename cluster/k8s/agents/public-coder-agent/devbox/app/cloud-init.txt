#cloud-config
write_files:
  - path: /etc/profile.d/public-coder-devbox-proxy.sh
    owner: root:root
    permissions: "0644"
    content: |
      export HTTP_PROXY=http://public-coder-agent-proxy.public-coder-agent.svc.cluster.local:8080
      export HTTPS_PROXY=http://public-coder-agent-proxy.public-coder-agent.svc.cluster.local:8080
      export http_proxy=http://public-coder-agent-proxy.public-coder-agent.svc.cluster.local:8080
      export https_proxy=http://public-coder-agent-proxy.public-coder-agent.svc.cluster.local:8080
      export NO_PROXY=127.0.0.1,localhost
      export no_proxy=127.0.0.1,localhost
      export GH_PAT=proxy-github-placeholder
      export SSL_CERT_FILE=/run/public-coder-devbox-proxy-ca/ca-bundle.crt
      export NIX_SSL_CERT_FILE=/run/public-coder-devbox-proxy-ca/ca-bundle.crt
      export CURL_CA_BUNDLE=/run/public-coder-devbox-proxy-ca/ca-bundle.crt
      export GIT_SSL_CAINFO=/run/public-coder-devbox-proxy-ca/ca-bundle.crt
runcmd:
  - |
    set -eu
    install -d -m 0700 /root/.ssh /run/public-coder-devbox-host-key/source
    mount -o ro /dev/disk/by-id/virtio-pchostkey /run/public-coder-devbox-host-key/source
    install -Dm0600 /run/public-coder-devbox-host-key/source/ssh_host_ed25519_key /etc/ssh/ssh_host_ed25519_key
    umount /run/public-coder-devbox-host-key/source
    key='ssh-ed25519 AAAAC3NzaC1lZDI1NTE5AAAAIIWU0hfZsf8lk6WB907Ppd49rQobJEsPhvzqGg2uvc1U public-coder-devbox OpenClaw root login'
    touch /root/.ssh/authorized_keys
    grep -qxF "$key" /root/.ssh/authorized_keys || printf '%s\n' "$key" >> /root/.ssh/authorized_keys
    chmod 0600 /root/.ssh/authorized_keys
    install -d -m 0755 /run/public-coder-devbox-proxy-ca/source
    mounted=0
    for _ in $(seq 1 60); do
      if mount -o ro /dev/disk/by-id/virtio-pcproxyca /run/public-coder-devbox-proxy-ca/source 2>/dev/null; then
        mounted=1
        break
      fi
      sleep 1
    done
    if [ "$mounted" -eq 1 ] && [ -s /run/public-coder-devbox-proxy-ca/source/ca-certificates.crt ]; then
      install -Dm0644 /run/public-coder-devbox-proxy-ca/source/ca-certificates.crt /run/public-coder-devbox-proxy-ca/proxy-ca.crt
      cat /etc/ssl/certs/ca-bundle.crt /run/public-coder-devbox-proxy-ca/proxy-ca.crt > /run/public-coder-devbox-proxy-ca/ca-bundle.crt
      umount /run/public-coder-devbox-proxy-ca/source
    else
      echo 'proxy CA ConfigMap disk not available yet; manual switch will need it' >&2
    fi
    mkdir -p /etc/nix
    cat > /etc/nix/nix.conf <<'EOF'
    experimental-features = nix-command flakes
    http-proxy = http://public-coder-agent-proxy.public-coder-agent.svc.cluster.local:8080
    ssl-cert-file = /run/public-coder-devbox-proxy-ca/ca-bundle.crt
    EOF
    systemctl try-restart sshd.service
